index.html

<!DOCTYPE html>
<html>
<body>

<h2>Cadastro</h2>

<form id="form">
<input id="nomeAluno" placeholder="Nome"><br>
<input id="idadeAluno" type="number" placeholder="Idade"><br>
<input id="cpfAluno" placeholder="CPF"><br>
<input id="instituicao" placeholder="Instituição"><br>

<select id="turno">
<option>Manhã</option>
<option>Tarde</option>
<option>Noite</option>
</select>

<button>Cadastrar</button>
</form>

<h3>Login</h3>
<input id="user">
<input id="pass" type="password">
<button onclick="login()">Entrar</button>

<ul id="lista"></ul>

<script>
const API = 'http://localhost:3000';
let token = '';

form.onsubmit = async e => {
  e.preventDefault();

  const data = {
    nomeAluno: nomeAluno.value,
    idadeAluno: parseInt(idadeAluno.value),
    cpfAluno: cpfAluno.value,
    instituicao: instituicao.value,
    turno: turno.value
  };

  await fetch(API + '/cadastro', {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify(data)
  });

  alert('Cadastrado!');
};

async function login() {
  const res = await fetch(API + '/login', {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify({user:user.value, pass:pass.value})
  });

  const j = await res.json();
  token = j.token;
  carregar();
}

async function carregar() {
  const res = await fetch(API + '/alunos', {
    headers: { Authorization: token }
  });

  const dados = await res.json();
  lista.innerHTML = '';

  dados.forEach(p => {
    const li = document.createElement('li');

    li.innerHTML = `
      ${p.nomeAluno} - ${p.instituicao}
      <button onclick="editar('${p._id}')">Editar</button>
      <button onclick="deletar('${p._id}')">Deletar</button>
    `;

    lista.appendChild(li);
  });
}

async function editar(id) {
  const nova = prompt('Nova instituição:');

  await fetch(API + '/aluno/' + id, {
    method: 'PUT',
    headers: {
      'Content-Type':'application/json',
      'Authorization':token
    },
    body: JSON.stringify({ instituicao: nova })
  });

  carregar();
}

async function deletar(id) {
  await fetch(API + '/aluno/' + id, {
    method: 'DELETE',
    headers: { Authorization: token }
  });

  carregar();
}
</script>

</body>
</html>
